```js
// ── Reminders (fuzzy windows + spaced-repetition bubble-up) ──────
// keel owns the mechanism; the /remind skill is the NL front-end. Pure here.

/**
 * @typedef {Object} Reminder
 * @property {string} id              short stable id, "r"+n
 * @property {string} text
 * @property {"approx"|"bubble"} kind
 * @property {number} createdTs
 * @property {number} nextTs          when it next becomes due to surface
 * @property {number} dismissedTs     0 = active; >0 = retired
 * @property {string} [ref]           reserved: secretariat envelope path (v2, unwired)
 * @property {number} [softTs]        approx: fuzzy target
 * @property {number} [slackDays]     approx: band half-width
 * @property {number} [box]           bubble: Leitner box index
 * @property {number} [lastRatedTs]   bubble: last rating time
 */

const DAY = 86_400_000;

/** Next free reminder id: "r" + (max existing numeric suffix + 1). @param {State} state */
export function nextReminderId(state) {
  const max = (state.reminders ?? []).reduce((m, r) => {
    const n = parseInt(String(r.id).replace(/^r/, ""), 10);
    return Number.isFinite(n) && n > m ? n : m;
  }, 0);
  return `r${max + 1}`;
}

/** Append a reminder. Pure. @param {State} state @param {Reminder} rem */
export function addReminder(state, rem) {
  return { ...state, reminders: [...(state.reminders ?? []), rem] };
}

/** @param {State} state @param {string} id @returns {Reminder|null} */
export const findReminder = (state, id) =>
  (state.reminders ?? []).find((r) => r.id === id) ?? null;
```

```js
/** Leitner ladder (days). got → box+1 · fuzzy → stay · miss → box 0. Long tail = sticks. */
export const BOXES = [1, 3, 7, 16, 35, 90, 180, 365];

/** A fresh spaced-repetition bubble: box 0, due in 1 day. @param {string} id @param {string} text @param {number} now */
export function newBubbleReminder(id, text, now) {
  return {
    id, text, kind: "bubble", createdTs: now,
    nextTs: now + BOXES[0] * DAY, dismissedTs: 0,
    box: 0, lastRatedTs: 0, ref: undefined,
  };
}

/** A fuzzy one-shot window. nextTs = band start. @param {string} id @param {string} text @param {number} softTs @param {number} slackDays @param {number} now */
export function newApproxReminder(id, text, softTs, slackDays, now) {
  return {
    id, text, kind: "approx", createdTs: now,
    softTs, slackDays, nextTs: softTs - slackDays * DAY,
    dismissedTs: 0, ref: undefined,
  };
}
```

```js
/** Apply a Leitner rating to a bubble by id. No-op on approx / unknown id. Pure.
 * @param {State} state @param {string} id @param {"got"|"fuzzy"|"miss"} grade @param {number} now */
export function rateReminder(state, id, grade, now) {
  const reminders = (state.reminders ?? []).map((r) => {
    if (r.id !== id || r.kind !== "bubble") return r;
    let box = r.box ?? 0;
    if (grade === "got") box = Math.min(box + 1, BOXES.length - 1);
    else if (grade === "miss") box = 0;
    // "fuzzy" → box unchanged
    return { ...r, box, nextTs: now + BOXES[box] * DAY, lastRatedTs: now };
  });
  return { ...state, reminders };
}

/** Retire a reminder (either kind) by id. Keeps the record, sets dismissedTs. Pure.
 * @param {State} state @param {string} id @param {number} now */
export function retireReminder(state, id, now) {
  const reminders = (state.reminders ?? []).map((r) =>
    r.id === id ? { ...r, dismissedTs: now } : r);
  return { ...state, reminders };
}
```

```js
const REMINDER_CAP = 5;
const bandEnd = (r) => r.softTs + r.slackDays * DAY;

/** Due reminder lines for the session-start nudge stack. Tiered + ordered + capped. Pure.
 * Order: overdue-approx → due-approx → due-bubble (most-overdue first within each). @param {State} state @param {number} now */
export function reminderLines(state, now) {
  const due = (state.reminders ?? []).filter((r) => !r.dismissedTs && r.nextTs <= now);
  const overdueApprox = [], dueApprox = [], dueBubble = [];
  for (const r of due) {
    if (r.kind === "approx") (now > bandEnd(r) ? overdueApprox : dueApprox).push(r);
    else if (r.kind === "bubble") dueBubble.push(r);
  }
  const byDue = (a, b) => a.nextTs - b.nextTs;
  const ordered = [...overdueApprox.sort(byDue), ...dueApprox.sort(byDue), ...dueBubble.sort(byDue)];
  const lines = ordered.slice(0, REMINDER_CAP).map((r) =>
    r.kind === "bubble"
      ? `[keel] 🧠 recall: ${r.text}  → keel remind got|fuzzy|miss ${r.id}`
      : now > bandEnd(r)
        ? `[keel] ⚠ overdue: ${r.text}  → keel remind done ${r.id}`
        : `[keel] ≈ due: ${r.text}  → keel remind done ${r.id}`);
  if (ordered.length > REMINDER_CAP)
    lines.push(`[keel] …+${ordered.length - REMINDER_CAP} more (keel remind list)`);
  return lines;
}

/** One line per active reminder for `keel remind list`. Pure. @param {State} state @param {number} now */
export function reminderListLines(state, now) {
  const active = (state.reminders ?? []).filter((r) => !r.dismissedTs);
  if (!active.length) return ["keel: no active reminders."];
  return active.map((r) => {
    const dueInD = Math.round((r.nextTs - now) / DAY);
    const when = dueInD <= 0 ? "due now" : `${dueInD}d`;
    return r.kind === "bubble"
      ? `${r.id} 🧠 box${r.box} next ${when} · ${r.text}`
      : `${r.id} ≈ approx next ${when} · ${r.text}`;
  });
}
```

```js
/** Parse `remind add` argv tail into { bubble, softTs, slackDays, text }. */
function parseAddArgs(rest) {
  const out = { bubble: false, softTs: 0, slackDays: 0, parts: [] };
  for (let i = 0; i < rest.length; i++) {
    const a = rest[i];
    if (a === "--bubble") out.bubble = true;
    else if (a === "--soft") out.softTs = Number(rest[++i]);
    else if (a === "--slack") out.slackDays = Number(rest[++i]);
    else out.parts.push(a);
  }
  out.text = out.parts.join(" ").trim();
  return out;
}

function cmdRemind(now, args) {
  const [sub, ...rest] = args;
  const state = loadState();

  if (sub === "add") {
    const a = parseAddArgs(rest);
    if (!a.text) { console.log('keel: remind add needs text. e.g. `keel remind add --bubble "<note>"`.'); return; }
    const id = nextReminderId(state);
    const rem = a.bubble
      ? newBubbleReminder(id, a.text, now)
      : newApproxReminder(id, a.text, a.softTs, a.slackDays, now);
    saveState(addReminder(state, rem));
    console.log(a.bubble
      ? `keel: bubble ${id} added — first recall in 1 day, then spaced out. \`keel remind got|fuzzy|miss ${id}\` when it surfaces.`
      : `keel: reminder ${id} added — surfaces around ${new Date(rem.softTs).toDateString()} (±${a.slackDays}d). \`keel remind done ${id}\` to clear.`);
    return;
  }

  if (sub === "list") { console.log(reminderListLines(state, now).join("\n")); return; }

  if (sub === "got" || sub === "fuzzy" || sub === "miss") {
    const id = rest[0];
    const r = findReminder(state, id);
    if (!r) { console.log(`keel: no reminder ${id}.`); return; }
    if (r.kind !== "bubble") { console.log(`keel: ${id} is an approx reminder — rate only bubbles. \`keel remind done ${id}\` to clear it.`); return; }
    const next = rateReminder(state, id, sub, now);
    saveState(next);
    const updated = findReminder(next, id);
    console.log(`keel: ${id} rated ${sub} — next recall in ${BOXES[updated.box]}d (box ${updated.box}).`);
    return;
  }

  if (sub === "done") {
    const id = rest[0];
    if (!findReminder(state, id)) { console.log(`keel: no reminder ${id}.`); return; }
    saveState(retireReminder(state, id, now));
    console.log(`keel: ${id} cleared.`);
    return;
  }

  console.log('keel remind: <add --bubble|--soft <ms> --slack <days> "<text>" | list | got|fuzzy|miss <id> | done <id>>');
}
```

# Remind

keel's fuzzy + spaced-repetition reminder front-end. You resolve the *language and the timing*; keel stores and surfaces. Hard calendar dates → Things; this is for what a calendar handles badly.

## Two modes

- **Approximate (`~`)** — a fuzzy one-shot window. The user named a soft time ("next week", "sometime in July"). Resolve it to a target + slack, store as `approx`. It nudges within a band, then escalates to overdue, until cleared.
- **Bubble-up (SRS)** — no date; the user wants to *remember / learn / internalize* something. Store as `bubble`. It resurfaces on a Leitner ladder (1 → 3 → 7 → 16 → 35 → 90 → 180 → 365 days), expanding on recall, resetting on a miss.

Pick by signal: a named time → approx. "remember / learn / don't let me forget / bubble up / resurface" → bubble.

## Add

**Bubble:**
```bash
node ~/.keel/keel.mjs remind add --bubble "<note>"
```

**Approximate** — resolve the phrase to `softTs` (epoch **ms**) and `slackDays`, then:
```bash
node ~/.keel/keel.mjs remind add --soft <epochMs> --slack <days> "<note>"
```

Resolution table (compute `softTs` from the current time; pick the row that fits, interpolate otherwise):

| Phrase | softTs | slackDays |
|---|---|---|
| "in a few days" | +3d | 2 |
| "next week" | +7d | 3 |
| "in a couple weeks" | +14d | 4 |
| "this month" / "end of the month" | last day of month | 5 |
| "sometime in <month>" | the 15th of that month | 15 |

Compute the epoch-ms with a one-liner if helpful, e.g.:
```bash
node -e 'console.log(Date.now() + 7*86400000)'   # next week
```
